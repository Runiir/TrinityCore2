cmake_minimum_required(VERSION 3.22)
project(client442_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(LIBEI_INCLUDE_DIR libei.h PATH_SUFFIXES libei-1.0 REQUIRED)
find_library(LIBEI_LIBRARY NAMES ei libei.so.1 REQUIRED)
add_executable(client442_input main.cpp ../native_bridge/json.cpp)
target_include_directories(client442_input PRIVATE ${LIBEI_INCLUDE_DIR} ../native_bridge)
target_link_libraries(client442_input PRIVATE ${LIBEI_LIBRARY})
target_compile_options(client442_input PRIVATE -Wall -Wextra -Wpedantic)
