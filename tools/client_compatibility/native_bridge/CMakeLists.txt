cmake_minimum_required(VERSION 3.22)
project(client442_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.5 REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_library(MYSQLCLIENT_LIBRARY mysqlclient REQUIRED)
add_library(bridge_protocol buffer.cpp json.cpp crypto.cpp fields.cpp protocol.cpp movement.cpp native_objects.cpp native_transport.cpp events.cpp equipment_sets.cpp appearance.cpp character_combat.cpp character_ratings.cpp
    object_fields.cpp public_players.cpp object_updates.cpp unit_updates.cpp player_rest.cpp player_ui_state.cpp reputation_fields.cpp inventory_requests.cpp inventory_updates.cpp glyph_fields.cpp banking.cpp inspection.cpp trading.cpp casting.cpp archaeology_casts.cpp archaeology_history.cpp extra_marker_casts.cpp auras.cpp initialization.cpp currency_requests.cpp looting.cpp transfers.cpp gossip.cpp taxi.cpp
    queries.cpp quests.cpp quest_status.cpp quest_list.cpp quest_progress.cpp quest_turnin.cpp quest_queries.cpp quest_rewards.cpp quest_fields.cpp talents.cpp mail.cpp mail_actions.cpp mail_send.cpp auctions.cpp auction_catalog.cpp auction_browse.cpp auction_item_search.cpp auction_transactions.cpp auction_data.cpp merchants.cpp repairs.cpp trainers.cpp item_notifications.cpp item_use.cpp creature_movement.cpp public_player_movement.cpp database.cpp data.cpp characters.cpp reputation.cpp social.cpp chat.cpp lifecycle.cpp guild_fields.cpp guild_packets.cpp guild_roster.cpp guild_membership.cpp guild_events.cpp guild_notes.cpp party.cpp party_requests.cpp party_state.cpp party_profiles.cpp party_roles.cpp raid_markers.cpp player_names_codec.cpp account_settings.cpp achievements.cpp realm_metadata.cpp)
target_include_directories(bridge_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_sources(bridge_protocol PRIVATE chat_probe.cpp chat_channels.cpp chat_ignore.cpp item_text.cpp who.cpp pet_packets.cpp pet_casts.cpp cast_packets.cpp melee_results.cpp control_skill_metadata.cpp pet_rename_probe.cpp pet_rename.cpp)
target_sources(bridge_protocol PRIVATE stable_probe.cpp stables.cpp)
target_sources(bridge_protocol PRIVATE spell_damage.cpp)
target_link_libraries(bridge_protocol PUBLIC OpenSSL::Crypto Threads::Threads ZLIB::ZLIB ${MYSQLCLIENT_LIBRARY})
target_compile_options(bridge_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(bridge_codec codec_main.cpp)
target_link_libraries(bridge_codec PRIVATE bridge_protocol)
add_library(bridge_runtime channel.cpp native.cpp service.cpp session_auth.cpp client_requests.cpp server_packets.cpp player_names.cpp marker_runtime.cpp ready_runtime.cpp)
target_sources(bridge_runtime PRIVATE who_runtime.cpp)
target_link_libraries(bridge_runtime PUBLIC bridge_protocol)
target_compile_options(bridge_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(client442_bridge main.cpp)
target_link_libraries(client442_bridge PRIVATE bridge_runtime)
enable_testing()
